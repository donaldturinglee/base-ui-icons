## Getting Started

### Installation

1. Install from npm
   ```sh
   npm install @gamecrafters/base-ui-icons
   ```

<p align="right">(<a href="#readme-top">back to top</a>)</p>



## Usage

```jsx
import { AddCircleRegular, AlertRegular } from '@gamecrafters/base-ui-icons'

const Toolbar = () => {
  return (
    <button type="button">
      <AddCircleRegular size={24} aria-label="Add item" />
    </button>
  )
}
```
